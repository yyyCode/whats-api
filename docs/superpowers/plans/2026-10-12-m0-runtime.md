```go
// Package model 是数据层：GORM 模型、连接管理、迁移、以及配置的持久化读写。
//
// 本项目要求同一份代码跑在 SQLite / MySQL / PostgreSQL 上，所以这里有两条纪律：
//  1. 不写数据库特有的语法，方言差异一律收进本文件的 helper；
//  2. 迁移必须幂等——连续启动三次不能有任何副作用。
package model

import (
	"context"
	"errors"
	"fmt"
	"strings"
	"time"

	"github.com/glebarez/sqlite"
	"gorm.io/driver/mysql"
	"gorm.io/driver/postgres"
	"gorm.io/gorm"
	"gorm.io/gorm/clause"
	gormlogger "gorm.io/gorm/logger"

	"github.com/yyyCode/whats-api/internal/common"
	"github.com/yyyCode/whats-api/internal/logger"
)

// DB 是主库；LOG_DB 是日志库（未单独配置时与 DB 是同一个连接）。
//
// 它们是"进程级单例句柄"：Init* 写一次，之后只读。测试通过重新 Init
// 指向临时库来注入——除此之外没有任何地方会写它们。
var (
	DB     *gorm.DB
	LOG_DB *gorm.DB
)

type dbKind string

const (
	dbKindSQLite   dbKind = "sqlite"
	dbKindMySQL    dbKind = "mysql"
	dbKindPostgres dbKind = "postgres"
)

// 连接池参数（仅用于非 SQLite：SQLite 是单写者，见 openDB）。
const (
	maxOpenConns    = 100
	maxIdleConns    = 20
	connMaxLifetime = time.Hour
	connMaxIdleTime = 10 * time.Minute
)

// InitDB 初始化主库并执行迁移。★这是唯一的致命初始化：
// 主库连不上就没有任何业务可谈，直接让启动失败（其余初始化失败都降级）。
func InitDB() error {
	dialector, kind, err := chooseDB(common.SQLDSN, false)
	if err != nil {
		return fmt.Errorf("主库 DSN 无效: %w", err)
	}

	db, err := openDB(dialector, kind)
	if err != nil {
		return fmt.Errorf("连接主库失败: %w", err)
	}
	DB = db

	if err := Migrate(); err != nil {
		return err
	}
	return nil
}

// InitLogDB 初始化日志库。调用点必须在 InitOptionMap 之后
// （日志库 DSN 将来可能来自配置项），并且 LOG_SQL_DSN 为空时复用主库连接。
func InitLogDB() error {
	if common.LogSQLDSN == "" || common.LogSQLDSN == common.SQLDSN {
		LOG_DB = DB
		logger.Info("未单独配置日志库，复用主库")
	} else {
		dialector, kind, err := chooseDB(common.LogSQLDSN, true)
		if err != nil {
			return fmt.Errorf("日志库 DSN 无效: %w", err)
		}
		logDB, err := openDB(dialector, kind)
		if err != nil {
			return fmt.Errorf("连接日志库失败: %w", err)
		}
		LOG_DB = logDB
		logger.Info("日志库已独立配置", "driver", LogDriver())
	}

	return migrateLOGDB()
}

// chooseDB 按 DSN 前缀选择驱动。返回 dialector 而不是连接：
// 连接由调用方建立，测试可以直接断言"分派对了"而不需要真实数据库。
func chooseDB(dsn string, isLog bool) (gorm.Dialector, dbKind, error) {
	switch {
	case dsn == "" || dsn == "local" || strings.HasPrefix(dsn, "local:"):
		return sqlite.Open(sqliteFile(dsn)), dbKindSQLite, nil

	case strings.HasPrefix(dsn, "postgres://"), strings.HasPrefix(dsn, "postgresql://"):
		// PreferSimpleProtocol 让驱动走简单查询协议，兼容 PgBouncer / Neon 这类连接池。
		return postgres.New(postgres.Config{DSN: dsn, PreferSimpleProtocol: true}), dbKindPostgres, nil

	case isClickHouseDSN(dsn):
		// 分支留好但明确不实现——比"假装支持然后连不上"诚实，
		// 也让配错 DSN 的人在启动日志里立刻看到原因。
		if !isLog {
			return nil, "", fmt.Errorf("主库不支持 ClickHouse（该 DSN 形式只允许用于日志库）: %s", dsn)
		}
		return nil, "", errors.New("ClickHouse 日志库尚未引入（计划在 M6）")

	default:
		return mysql.Open(ensureParseTime(dsn)), dbKindMySQL, nil
	}
}

// isClickHouseDSN 识别 ClickHouse 的几种 DSN 写法（驱动尚未引入）。
func isClickHouseDSN(dsn string) bool {
	for _, prefix := range []string{"clickhouse://", "tcp://", "http://", "https://"} {
		if strings.HasPrefix(dsn, prefix) {
			return true
		}
	}
	return false
}

// sqliteFile 把 DSN 解析成 SQLite 文件路径。
func sqliteFile(dsn string) string {
	switch {
	case dsn == "", dsn == "local", dsn == "local:":
		return common.DefaultSQLiteFile
	default:
		if path := strings.TrimSpace(strings.TrimPrefix(dsn, "local:")); path != "" {
			return path
		}
		return common.DefaultSQLiteFile
	}
}

// ensureParseTime 给 MySQL DSN 补 parseTime=true。
// 缺了它，DATETIME/TIMESTAMP 列会以 []byte 返回，扫进 time 类型时直接报错。
func ensureParseTime(dsn string) string {
	if strings.Contains(dsn, "parseTime=") {
		return dsn
	}
	sep := "?"
	if strings.Contains(dsn, "?") {
		sep = "&"
	}
	return dsn + sep + "parseTime=true"
}

func openDB(dialector gorm.Dialector, kind dbKind) (*gorm.DB, error) {
	db, err := gorm.Open(dialector, &gorm.Config{
		Logger: gormLogger(),
		// 不让 AutoMigrate 建外键约束：三库对 ON DELETE 行为不一致，
		// 而本项目的一致性不靠数据库约束保证（日志与账本更是只追加）。
		DisableForeignKeyConstraintWhenMigrating: true,
	})
	if err != nil {
		return nil, err
	}

	sqlDB, err := db.DB()
	if err != nil {
		return nil, err
	}

	if kind == dbKindSQLite {
		// SQLite 是单写者：连接数放开只会制造 "database is locked"。
		// 限制为 1 之后，写操作在 Go 侧排队，比在 SQLite 侧互相报错好。
		sqlDB.SetMaxOpenConns(1)
		sqlDB.SetMaxIdleConns(1)
		if err := applySQLitePragmas(db); err != nil {
			return nil, err
		}
	} else {
		sqlDB.SetMaxOpenConns(maxOpenConns)
		sqlDB.SetMaxIdleConns(maxIdleConns)
		sqlDB.SetConnMaxLifetime(connMaxLifetime)
		sqlDB.SetConnMaxIdleTime(connMaxIdleTime)
	}

	return db, nil
}

// applySQLitePragmas 设置 SQLite 的行为。三条都不是可选项：
//   - WAL：读不阻塞写。多实例共用一个文件时，没有 WAL 就是随机 "database is locked"；
//   - busy_timeout：遇到写锁先等 5 秒再报错，而不是立刻失败（多实例的抖动容忍）；
//   - foreign_keys：SQLite 默认**关闭**外键约束，与 MySQL/PG 行为不一致。
func applySQLitePragmas(db *gorm.DB) error {
	for _, pragma := range []string{
		"PRAGMA journal_mode=WAL",
		"PRAGMA busy_timeout=5000",
		"PRAGMA foreign_keys=ON",
	} {
		if err := db.Exec(pragma).Error; err != nil {
			return fmt.Errorf("执行 %s 失败: %w", pragma, err)
		}
	}
	return nil
}

func gormLogger() gormlogger.Interface {
	if common.GinMode == "debug" {
		return gormlogger.Default.LogMode(gormlogger.Info)
	}
	// 生产模式只记慢查询与错误：把每条 SQL 刷进日志会把有用的行淹掉。
	return gormlogger.Default.LogMode(gormlogger.Warn)
}

// CloseDB 关闭主库与日志库（同一个连接时只关一次）。
// 停机时序里它必须是最后一步：早于后台 goroutine 退出就会让它们写到已关闭的连接上。
func CloseDB() error {
	var errs []error

	if LOG_DB != nil && LOG_DB != DB {
		if sqlDB, err := LOG_DB.DB(); err == nil {
			errs = append(errs, sqlDB.Close())
		}
	}
	if DB != nil {
		if sqlDB, err := DB.DB(); err == nil {
			errs = append(errs, sqlDB.Close())
		}
	}

	DB, LOG_DB = nil, nil
	return errors.Join(errs...)
}

// ---------- 方言 helper ----------
//
// 这些函数在 M0 只有测试调用（M0 没有裸 SQL，GORM 自己会给标识符加引号）。
// 它们存在的意义是：M1/M2 写条件 UPDATE 与裸 SQL 时不必回头改数据层约定。
// 实现为函数而不是包级变量——方言要等 InitDB 之后才知道，
// 做成变量会在初始化早期被读到空串，静默出错。

func dialectName() string {
	if DB == nil {
		return ""
	}
	return DB.Dialector.Name()
}

func isSQLite() bool   { return dialectName() == string(dbKindSQLite) }
func isPostgres() bool { return dialectName() == string(dbKindPostgres) }
func isMySQL() bool    { return dialectName() == string(dbKindMySQL) }

// commonKeyCol 返回保留字列 key 的引用写法（tokens.key、channels.key、options.key）。
// 三库里 key 都是关键字，裸写会报语法错误。
func commonKeyCol() string { return keyColFor(dialectName()) }

func keyColFor(dialect string) string {
	if dialect == string(dbKindPostgres) {
		return `"key"`
	}
	return "`key`" // MySQL / SQLite
}

// commonTrueVal / commonFalseVal 返回当前方言的布尔字面量。
// 裸 SQL 里写 1/0 在 MySQL 能用，在 PostgreSQL 的 boolean 列上会报类型错误。
func commonTrueVal() string  { return trueValFor(dialectName()) }
func commonFalseVal() string { return falseValFor(dialectName()) }

func trueValFor(dialect string) string {
	if dialect == string(dbKindPostgres) {
		return "true"
	}
	return "1"
}

func falseValFor(dialect string) string {
	if dialect == string(dbKindPostgres) {
		return "false"
	}
	return "0"
}

// lockForUpdate 给查询加行锁（M2 的并发预扣会用到）。
//
// ★ 绝不要写 tx.Set("gorm:query_option", "FOR UPDATE")：GORM v2 会**静默忽略**它，
// 于是代码看起来加了锁、SQL 里却没有，这是并发保护里最阴险的一类 bug。
func lockForUpdate(tx *gorm.DB) *gorm.DB { return lockForUpdateFor(dialectName(), tx) }

func lockForUpdateFor(dialect string, tx *gorm.DB) *gorm.DB {
	if dialect == string(dbKindSQLite) {
		// SQLite 是单写者 + 库级写锁，没有行锁语法；原样返回而不是报错。
		return tx
	}
	return tx.Clauses(clause.Locking{Strength: "UPDATE"})
}

// PingMain / PingLog 供 /api/status 的 ok 字段。带超时：
// 探活接口不能被一个卡住的连接拖住。
func PingMain() bool { return ping(DB) }
func PingLog() bool  { return ping(LOG_DB) }

func ping(db *gorm.DB) bool {
	if db == nil {
		return false
	}
	sqlDB, err := db.DB()
	if err != nil {
		return false
	}
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	defer cancel()
	return sqlDB.PingContext(ctx) == nil
}

// DSNKind 描述连接去向，用于 /api/status 展示。
// 只回一个枚举值，不回显 DSN 全文——DSN 里有密码。
func DSNKind() string {
	dsn := common.SQLDSN
	switch {
	case strings.HasPrefix(dsn, "postgres://"), strings.HasPrefix(dsn, "postgresql://"):
		return "server"
	case dsn != "" && !strings.HasPrefix(dsn, "local"):
		return "server" // MySQL
	case strings.Contains(dsn, ":memory:"):
		return "memory"
	default:
		return "file"
	}
}

	return nil

		logger.Info("非 master 节点，跳过主库迁移", "node_type", common.NodeType)
		return nil
	}
	if DB == nil {
		return errors.New("主库未初始化")
	}

	logger.Info("开始主库迁移", "tables", len(mainModels()))
	if err := DB.AutoMigrate(mainModels()...); err != nil {
		return fmt.Errorf("主库 AutoMigrate 失败: %w", err)
	}
	logger.Info("主库迁移完成")
	return nil
}

// migrateLOGDB 迁移日志库。LOG_SQL_DSN 为空时 LOG_DB == DB，
// 等于把 logs 建在主库里——单机部署的合理简化。
func migrateLOGDB() error {
	if !common.IsMasterNode {
		return nil
	}
	if LOG_DB == nil {
		return errors.New("日志库未初始化")
	}

	if err := LOG_DB.AutoMigrate(&Log{}); err != nil {
		return fmt.Errorf("日志库 AutoMigrate 失败: %w", err)
	}
	return nil
}

package model

import (
	"context"
	"fmt"
	"time"

	"gorm.io/gorm/clause"

	"github.com/yyyCode/whats-api/internal/logger"
	"github.com/yyyCode/whats-api/internal/setting/config"
	"github.com/yyyCode/whats-api/internal/setting/system_setting"
)

// Option 是运行时配置的持久化形态：key → JSON 编码的值。
//
// 表里只存"管理员显式改过"的项，默认值来自代码里的注册结构体
// （见 internal/setting/config）。这样将来调默认值能影响所有未显式修改的实例。
type Option struct {
	// key 是三库的关键字，列名必须显式指定；GORM 生成的 DDL 会给它加引号。
	Key   string `gorm:"column:key;primaryKey;size:128"`
	Value string `gorm:"type:text"`
}

// OptionMap 是内存中的配置镜像：key → JSON 编码的值。
//
// 它存在的意义是"比较基准"：SyncOptions 每轮把库里的行与它对比，
// 只对真的变了的项走一遍应用流程（否则每轮都在重建快照、刷日志）。
//
// 它是进程级可变状态，但只有 updateOptionMap 一个写入点。
var OptionMap = map[string]string{}

// InitOptionMap 装载配置：先用注册表里的默认值打底，再用库里的行覆盖。
//
// ★ 必须在任何读配置的模块之前调用（设计文档 §3 的第 4 步）——
// 包括 InitLogDB（日志库 DSN 将来可能来自配置）。
func InitOptionMap() error {
	OptionMap = config.GlobalConfig.ExportAllConfigs()

	rows, err := loadOptionsFromDatabase()
	if err != nil {
		return err
	}

	for key, value := range rows {
		if err := updateOptionMap(key, value); err != nil {
			// 库里的脏数据（比如降级安装残留的 key）不该让服务起不来。
			// 跳过并告警，等管理员来修——注意此时 OptionMap 保持默认值。
			logger.Warn("忽略无法应用的配置项", "key", key, "value", value, "error", err)
		}
	}

	logger.Info("配置已装载", "keys", len(OptionMap), "from_db", len(rows))
	return nil
}

// UpdateOption 是配置的**唯一写入路径**。
//
// 顺序不能颠倒：校验 → 落库 → 改内存。
// 先改内存再落库的话，落库失败会留下"这个实例和库不一致"的状态，
// 重启后表现为"配置莫名其妙回退了"，而且排查时看不到任何报错。
func UpdateOption(key, value string) error {
	// 校验在落库之前完成：库里的脏值会被别的实例同步过去，一处错变成处处错。
	if err := config.GlobalConfig.ValidateConfig(key, value); err != nil {
		return err
	}
	if err := saveOption(key, value); err != nil {
		return err
	}
	if err := updateOptionMap(key, value); err != nil {
		return err
	}
	logger.Info("配置已更新", "key", key)
	return nil
}

// updateOptionMap 是唯一的配置分发点：内存 map、运行时结构体、副作用三者一起改。
//
// 分成三件事的原因：
//   - OptionMap 是同步比较基准；
//   - UpdateConfig 写注册结构体并触发 OnRebuild，从而**原子发布**该域的快照；
//   - applySideEffects 处理没有对应结构体字段的开关（M0 只有日志级别）。
func updateOptionMap(key, value string) error {
	if err := config.GlobalConfig.UpdateConfig(key, value); err != nil {
		return err
	}
	OptionMap[key] = value
	applySideEffects(key)
	return nil
}

// applySideEffects 处理"非结构化副作用"，集中在这一个 switch 里。
//
// M0 只有日志级别需要它：其余配置项改完由读快照的代码（中间件、/api/status）自动生效，
// 不需要在这里加任何东西——每加一条都是一个新的隐式耦合点。
func applySideEffects(key string) {
	switch key {
	case "system_setting.log_level":
		level := system_setting.Get().LogLevel
		logger.SetLevel(logger.ParseLevel(level))
		logger.Info("日志级别已更新", "log_level", level)
	}
}

// saveOption 把单个配置项 upsert 进库。
//
// clause.OnConflict 让 GORM 按方言生成 MySQL 的 ON DUPLICATE KEY UPDATE
// 或 PostgreSQL / SQLite 的 ON CONFLICT DO UPDATE——一份代码，三库通用。
func saveOption(key, value string) error {
	row := Option{Key: key, Value: value}
	err := DB.Clauses(clause.OnConflict{
		Columns:   []clause.Column{{Name: "key"}},
		DoUpdates: clause.AssignmentColumns([]string{"value"}),
	}).Create(&row).Error
	if err != nil {
		return fmt.Errorf("保存配置 %q 失败: %w", key, err)
	}
	return nil
}

func loadOptionsFromDatabase() (map[string]string, error) {
	var rows []Option
	if err := DB.Find(&rows).Error; err != nil {
		return nil, fmt.Errorf("读取 options 表失败: %w", err)
	}

	out := make(map[string]string, len(rows))
	for _, r := range rows {
		out[r.Key] = r.Value
	}
	return out, nil
}

// SyncOptions 定时把库里的配置重新装载一遍，直到 ctx 被取消。
//
// **全量重载**而不是增量：某实例错过一次变更（网络抖动、进程重启）后，
// 下一轮就能自己追上，不需要任何"补偿"逻辑。
// 但只对值真的变了的项走应用流程，避免无谓的快照重建与日志噪声。
func SyncOptions(ctx context.Context, freq time.Duration) {
	ticker := time.NewTicker(freq)
	defer ticker.Stop()

	logger.Info("配置定时同步已启动", "interval_seconds", int(freq.Seconds()))
	for {
		select {
		case <-ctx.Done():
			logger.Info("配置定时同步已停止")
			return
		case <-ticker.C:
			applied, err := syncOptionsOnce()
			if err != nil {
				// 一轮失败不是事故：下一轮会重试，全量重载保证最终收敛。
				logger.Warn("配置同步失败，下一轮重试", "error", err)
				continue
			}
			if applied > 0 {
				logger.Info("配置同步完成", "applied", applied)
			}
		}
	}
}

// syncOptionsOnce 执行一轮同步，返回本轮实际应用的项数（测试据此断言"没变就不应用"）。
func syncOptionsOnce() (int, error) {
	pending, err := loadOptionsFromDatabase()
	if err != nil {
		return 0, err
	}

	applied := 0
	for key, value := range pending {
		if OptionMap[key] == value {
			continue
		}
		if err := updateOptionMap(key, value); err != nil {
			logger.Warn("同步配置项失败，跳过", "key", key, "error", err)
			continue
		}
		applied++
	}
	return applied, nil
}

// Package middleware 是全局中间件。
//
// 依赖规则：本包不 import controller（避免与路由层成环）。
// 因此这里的错误响应是手写的 JSON 字面量，而不是复用 controller 的信封函数——
// 这是一处刻意的重复：三个字段的字面量重复一次，好过让中间件依赖 handler 层。
package middleware

import (
	"crypto/rand"
	"encoding/hex"

	"github.com/gin-gonic/gin"
)

const (
	// RequestIDKey 是 gin.Context 里的键名。
	RequestIDKey = "request_id"
	// RequestIDHeader 是入站沿用、出站写回的响应头。
	RequestIDHeader = "X-Request-Id"
)

// RequestId 保证每个请求都有一个可追踪的 ID：
// 入站带了就沿用（便于跨服务串联），没带就生成。
// 没有它，线上排查只能靠时间戳对日志。
func RequestId() gin.HandlerFunc {
	return func(c *gin.Context) {
		id := c.GetHeader(RequestIDHeader)
		if id == "" {
			id = generateRequestID()
		}
		c.Set(RequestIDKey, id)
		c.Header(RequestIDHeader, id)
		c.Next()
	}
}

// GetRequestID 读取当前请求的 ID；缺失时返回空串（不 panic）。
func GetRequestID(c *gin.Context) string {
	if v, ok := c.Get(RequestIDKey); ok {
		if s, ok := v.(string); ok {
			return s
		}
	}
	return ""
}

// generateRequestID 生成 128 位随机 ID（32 位十六进制）。
// 用 crypto/rand 而不是 math/rand：ID 会出现在日志与响应头里，
// 可预测的 ID 意味着可以被用来伪造/关联他人请求。
func generateRequestID() string {
	buf := make([]byte, 16)
	if _, err := rand.Read(buf); err != nil {
		// crypto/rand 失败是系统级异常；退回空串让上层拒绝而不是用弱随机数冒充。
		return ""
	}
	return hex.EncodeToString(buf)
}
```

package middleware

import (
	"strings"

	"github.com/gin-gonic/gin"
)

// 三个面：转发面 / 管理面 / 其他（静态资源与 SPA）。
const (
	RouteTagRelay = "relay"
	RouteTagAPI   = "api"
	RouteTagWeb   = "web"

	// RouteTagKey 是 gin.Context 里的键名。
	RouteTagKey = "route_tag"
)

// RouteTag 按路径前缀给请求打面标签，供访问日志按面聚合。
//
// 为什么用全局中间件而不是挂在路由组上：M0 还没有真正的 /v1 端点，
// 按组挂会得到一个没有 handler 的组、也拿不到标签。
// M1 有了 /v1 handler 后，若需要更细的标签（如 /v1/tasks 与 /v1/chat 分开），
// 改成按组挂即可——AccessLog 读的是同一个 context key，不用动。
func RouteTag() gin.HandlerFunc {
	return func(c *gin.Context) {
		c.Set(RouteTagKey, tagForPath(c.Request.URL.Path))
		c.Next()
	}
}

func tagForPath(path string) string {
	switch {
	case path == "/v1" || strings.HasPrefix(path, "/v1/"):
		return RouteTagRelay
	case path == "/api" || strings.HasPrefix(path, "/api/"):
		return RouteTagAPI
	default:
		return RouteTagWeb
	}
}

// GetRouteTag 读取当前请求的面标签。
func GetRouteTag(c *gin.Context) string {
	if v, ok := c.Get(RouteTagKey); ok {
		if s, ok := v.(string); ok {
			return s
		}
	}
	return RouteTagWeb
}

package middleware

import (
	"fmt"
	"net/http"
	"runtime/debug"

	"github.com/gin-gonic/gin"

	"github.com/yyyCode/whats-api/internal/logger"
)

// CustomRecovery 捕获 panic：记结构化日志（含堆栈），返回统一 JSON 500。
//
// 两个刻意的决定：
//  1. 堆栈只进日志、不进响应——堆栈里有内部路径与变量值，属于信息泄漏；
//  2. 已经开始写响应时（未来是 SSE 流）不再改状态码，只能中断连接——
//     往半截流里再写一段 JSON 只会让客户端更难解析。
func CustomRecovery() gin.HandlerFunc {
	return func(c *gin.Context) {
		defer func() {
			r := recover()
			if r == nil {
				return
			}

			logger.Error("请求处理 panic",
				"panic", fmt.Sprint(r),
				"stack", string(debug.Stack()),
				"method", c.Request.Method,
				"path", c.Request.URL.Path,
				"route_tag", GetRouteTag(c),
				"request_id", GetRequestID(c),
			)

			if c.Writer.Written() {
				c.Abort()
				return
			}
			// /api 的信封（与 controller/response.go 的 APIResponse 同形）。
			// 这里手写而不是调用 controller：中间件不依赖 handler 层，避免成环。
			c.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{
				"success": false,
				"message": "服务器内部错误",
				"data":    nil,
			})
		}()

		c.Next()
	}
}

package middleware

import (
	"net/http"
	"time"

	"github.com/gin-gonic/gin"

	"github.com/yyyCode/whats-api/internal/cachex"
	"github.com/yyyCode/whats-api/internal/logger"
	"github.com/yyyCode/whats-api/internal/setting/system_setting"
)

// TotalCounterKey 是请求总数计数器的键名。
const TotalCounterKey = "total"

// AccessLog 输出结构化访问日志，并累加请求总数。
//
// 配置读取走 system_setting.Get() 这份**不可变快照**，不读可写结构体：
// 直接读可写实例会在热更新时产生 data race（设计文档 §4.3）。
//
// 计数器为 nil 时照常工作（单测与极端初始化顺序下不该因为一个 nil 崩掉）。
func AccessLog(requests *cachex.Counter) gin.HandlerFunc {
	return func(c *gin.Context) {
		start := time.Now()
		path := c.Request.URL.Path

		c.Next()

		// 计数与日志是两件事：关掉访问日志后计数仍然要涨，
		// 否则 /api/status 的 requests_total 会随日志开关一起"假装没流量"。
		if requests != nil {
			_, _ = requests.Incr(TotalCounterKey)
		}

		cfg := system_setting.Get()
		if cfg == nil || !cfg.EnableAccessLog {
			return
		}

		latency := time.Since(start)
		attrs := []any{
			"method", c.Request.Method,
			"path", path,
			"status", c.Writer.Status(),
			"latency_ms", latency.Milliseconds(),
			"size", c.Writer.Size(),
			"route_tag", GetRouteTag(c),
			"request_id", GetRequestID(c),
			"client_ip", c.ClientIP(),
		}
		if q := c.Request.URL.RawQuery; q != "" {
			attrs = append(attrs, "query", q)
		}
		// 慢请求标记：运维筛日志时"先看 slow"比按延迟排序方便。
		if cfg.SlowRequestThresholdSeconds > 0 && latency.Seconds() >= cfg.SlowRequestThresholdSeconds {
			attrs = append(attrs, "slow", true)
		}

		switch status := c.Writer.Status(); {
		case status >= http.StatusInternalServerError:
			logger.Error("请求完成", attrs...)
		case status >= http.StatusBadRequest:
			logger.Warn("请求完成", attrs...)
		default:
			logger.Info("请求完成", attrs...)
		}
	}
}

```go
package middleware

import (
	"net/http"

	"github.com/gin-gonic/gin"

	"github.com/yyyCode/whats-api/internal/common"
)

// CORS 按 CORS_ALLOW_ORIGINS 处理跨域。
//
// M0 不设置 Access-Control-Allow-Credentials：鉴权在 M1 引入，
// 届时若要用 Cookie 承载会话，必须改成"回显具体 Origin"——
// 通配符 * 与 credentials 在浏览器里不能共存，混用会让请求被静默拒绝。
func CORS() gin.HandlerFunc {
	allowAll := len(common.CORSAllowOrigins) == 1 && common.CORSAllowOrigins[0] == "*"
	allowed := make(map[string]bool, len(common.CORSAllowOrigins))
	for _, origin := range common.CORSAllowOrigins {
		allowed[origin] = true
	}

	return func(c *gin.Context) {
		origin := c.GetHeader("Origin")

		switch {
		case origin == "":
			// 非浏览器请求（curl / SDK / 探活）：没有 Origin，不需要也不该有 CORS 头
		case allowAll:
			c.Header("Access-Control-Allow-Origin", "*")
		case allowed[origin]:
			c.Header("Access-Control-Allow-Origin", origin)
			// 按来源变化的响应必须声明 Vary，否则中间缓存会把 A 源的响应发给 B 源
			c.Header("Vary", "Origin")
		}

		if c.Request.Method == http.MethodOptions {
			c.Header("Access-Control-Allow-Methods", "GET, POST, PUT, PATCH, DELETE, OPTIONS")
			c.Header("Access-Control-Allow-Headers", "Authorization, Content-Type, X-Request-Id")
			c.Header("Access-Control-Max-Age", "86400")
			c.AbortWithStatus(http.StatusNoContent)
			return
		}

		c.Next()
	}
}

// Package controller 是 HTTP handler 层。
//
// /api（管理面）与 /v1（转发面）的错误形状从第一天就不同：
// 管理面用这里的 {success, message, data} 信封（将来 message 旁边加 message_key 供 i18n），
// 转发面将来是 OpenAI/Claude 形状的错误 JSON。两套形状各自演进，不互相将就。
package controller

import (
	"net/http"

	"github.com/gin-gonic/gin"
)

// APIResponse 是管理面的统一信封。
type APIResponse struct {
	Success bool   `json:"success"`
	Message string `json:"message"`
	Data    any    `json:"data"`
}

// APISuccess 返回成功信封。data 为 nil 时是合法的"无数据成功"。
func APISuccess(c *gin.Context, data any) {
	c.JSON(http.StatusOK, APIResponse{Success: true, Message: "", Data: data})
}

// APIError 返回失败信封。message 是给人读的；
// 将来接 i18n 时在旁边加 message_key，而不是改 message 的内容。
func APIError(c *gin.Context, status int, message string) {
	c.JSON(status, APIResponse{Success: false, Message: message, Data: nil})
}

//
// 计数器用接口而不是 *cachex.Counter：
//   - 依赖规则里 controller 直连 cachex 是一条多余的边（它经由 model 已经可达）；
//   - 单测里可以塞一个假计数器，不必真起缓存。
//

	Requests  Counter
	StartedAt time.Time

	Get(key string) (int64, bool)

//
// 这里**不含任何敏感信息**：没有 DSN 全文（DSN 里有密码）、没有密钥、

	LogLevel                    string  `json:"log_level"`

	SlowRequestThresholdSeconds float64 `json:"slow_request_threshold_seconds"`

	RegisteredKeys int      `json:"registered_keys"`

// Status 是 M0 唯一的业务端点：既是探活，也是验收的观察窗。
//
// 管理面的写接口是 M3 的事，所以"改配置不重启即生效"这条验收标准
// 必须在 /api/status 上可被观察到——它不是调试信息，是验收装置。
func Status(deps Deps) gin.HandlerFunc {
	return func(c *gin.Context) {
		APISuccess(c, buildStatus(deps))
	}
}

func buildStatus(deps Deps) StatusData {
	cfg := system_setting.Get()

	// 探活失败不是错误：接口本身仍要返回 200，
	// 让运维能看到"服务活着但库连不上"这个事实（用 500 反而看不出是哪一段坏了）。
	setupComplete, _ := model.SetupComplete()

	data := StatusData{
		Version:       common.Version,
		UptimeSeconds: int64(time.Since(deps.StartedAt).Seconds()),
		NodeType:      common.NodeType,
		NodeName:      common.NodeName,
		IsMaster:      common.IsMasterNode,

		DB: DBStatus{
			Driver:  model.MainDriver(),
			DSNKind: model.DSNKind(),
			OK:      model.PingMain(),
		},
		LogDB: LogDBStatus{
			Driver:         model.LogDriver(),
			SharedWithMain: !model.LogDBSeparate(),
			OK:             model.PingLog(),
		},
		Cache: CacheStatus{
			Backend:         cacheBackend(deps),
			RedisConfigured: common.RedisConnString != "",
		},
		Config: ConfigStatus{
			Modules:        config.GlobalConfig.Modules(),
			RegisteredKeys: config.GlobalConfig.FieldCount(),
		},

		SetupComplete: setupComplete,
	}

	if cfg != nil {
		data.StatusMessage = cfg.StatusMessage
		data.LogLevel = cfg.LogLevel
		data.EnableAccessLog = cfg.EnableAccessLog
		data.SlowRequestThresholdSeconds = cfg.SlowRequestThresholdSeconds
	}

	if deps.Requests != nil {
		data.RequestsTotal, _ = deps.Requests.Get("total")
	}

	return data
}

// cacheBackend 报告实际生效的缓存后端。
// 读 RedisEnabled 而不是"是否配置了 REDIS_CONN_STRING"：
// 配了但连不上时应当显示 memory——那才是实际发生的事。
func cacheBackend(deps Deps) string {
	if common.RedisEnabled {
		return "redis"
	}
	return "memory"
}

// Package router 是路由注册层。
//
// 三层分离（照 02 篇）：/api 管理面、/v1 转发面、以及兜底的 web/NoRoute。
// 分离的意义不只是整洁：两个面的错误形状、中间件、鉴权方式、SLA 都不同，
// 混在一个组里，将来给 /v1 加鉴权就会误伤管理面。
package router

import (
	"time"

	"github.com/gin-gonic/gin"

	"github.com/yyyCode/whats-api/internal/cachex"
	"github.com/yyyCode/whats-api/internal/controller"
	"github.com/yyyCode/whats-api/internal/middleware"
)

// Deps 是路由层需要的进程级依赖。
type Deps struct {
	// Requests 是请求总数计数器（/api/status 暴露）。
	Requests *cachex.Counter
	// StartedAt 是进程启动时间，用来算 uptime。
	StartedAt time.Time
}

// SetRouter 注册全部路由与全局中间件。
//
// 中间件顺序是有讲究的（设计文档 §7.1）：
// recovery 最外层（panic 也要被日志与统一响应覆盖）→ request-id（后续所有日志都要带它）
// → version → route-tag → access-log（此时前几层都注入了，日志才完整）→ cors（最内层，预检可提前 abort）。
func SetRouter(engine *gin.Engine, deps Deps) {
	// 不用 gin.Default()：它自带的 Logger 会往 stderr 打一份非结构化日志，
	// 与自研访问日志重复；用 gin.New() + 自己的中间件。
	engine.Use(
		middleware.CustomRecovery(),
		middleware.RequestId(),
		middleware.Version(),
		middleware.RouteTag(),
		middleware.AccessLog(deps.Requests),
		middleware.CORS(),
	)

	cdeps := controller.Deps{Requests: deps.Requests, StartedAt: deps.StartedAt}

package router

import (
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"

	"github.com/yyyCode/whats-api/internal/controller"
)

// setWebRouter 注册兜底路由。
//
// 兜底必须是 JSON 404，而不是"所有未知路径都返回前端 HTML"：
// 对于 API 路径，被吞成 200 + HTML 是极难排查的一类故障
// （SDK 报"解析响应失败"，而不是"404 路径不存在"）。
// 前端 SPA 兜底在 M3 引入时，也只对**明确的静态资源与非 API 路径**生效。
func setWebRouter(engine *gin.Engine) {
	engine.NoRoute(func(c *gin.Context) {
		path := c.Request.URL.Path

		switch {
		case path == "/v1" || strings.HasPrefix(path, "/v1/"):
			// 转发面的错误形状与 /api 不同：将来是 OpenAI/Claude 形状的 error 对象。
			// M0 先给出同形状的明确 JSON，让客户端不必猜。
			c.JSON(http.StatusNotFound, gin.H{
				"error": gin.H{
					"message": "转发端点尚未实现（计划在 M1）",
					"type":    "invalid_request_error",
					"code":    "not_implemented",
				},
			})

		case path == "/api" || strings.HasPrefix(path, "/api/"):
			controller.APIError(c, http.StatusNotFound, "接口不存在")

		case strings.HasPrefix(path, "/assets/"):
			controller.APIError(c, http.StatusNotFound, "静态资源不存在")

		default:
			controller.APIError(c, http.StatusNotFound, "前端资源尚未构建（计划在 M3）")
		}
	})
}

```go
// Package app 是进程的生命周期：启动时序、HTTP 服务、优雅停机。
//
// 为什么把启动做成 Run(ctx) 而不是写在 main() 里：
// 停机路径只有能被"直接取消 ctx"驱动，才是可测试的。
// 本机（Windows）无法向原生进程投递 SIGTERM，把信号处理留在 main.go 做一层薄壳，
// 这里就能真实测到"在途请求跑完 + DB 关闭 + 后台 goroutine 退出"。
package app

import (
	"context"
	"errors"
	"fmt"
	"net/http"
	"sync"
	"time"

	"github.com/gin-gonic/gin"
	"github.com/joho/godotenv"

	"github.com/yyyCode/whats-api/internal/cachex"
	"github.com/yyyCode/whats-api/internal/common"
	"github.com/yyyCode/whats-api/internal/logger"
	"github.com/yyyCode/whats-api/internal/model"
	"github.com/yyyCode/whats-api/internal/router"
	"github.com/yyyCode/whats-api/internal/setting/system_setting"
)

// requestsCounterTTL 是请求计数器的滚动窗口。
// 用 24 小时而不是进程生命周期：重启后计数不该从 0 开始"假装没流量"，
// 但也要有个终点的，否则键永远不失效。
const requestsCounterTTL = 24 * time.Hour

// Options 是 RunWithOptions 的可注入项。
type Options struct {
	// InjectRoutes 在 SetRouter 之后追加路由。
	//
	// 存在的唯一理由是测试：验收标准④要求"在途请求跑完"，而 M0 没有任何
	// 慢到能观测到停机排水的生产端点。生产代码永远不传它。
	InjectRoutes func(*gin.Engine)
}

// Run 启动服务直到 ctx 被取消，然后优雅停机。
func Run(ctx context.Context) error { return RunWithOptions(ctx, Options{}) }

// RunWithOptions 是 Run 的可注入版本。
func RunWithOptions(ctx context.Context, opts Options) error {
	// ---------- ① 环境变量与日志 ----------
	// godotenv 只补"进程环境里还没有"的变量，因此容器里注入的环境变量优先，
	// .env 只是本地开发的便利。
	dotenvErr := godotenv.Load(".env")

	common.InitEnv()
	logger.SetupLogger(common.LogLevel, false)

	if dotenvErr != nil {
		logger.Debug("未加载 .env 文件，使用进程环境变量", "error", dotenvErr)
	}
	// InitEnv 跑在 logger 之前，告警先攒着，现在统一输出。
	for _, warning := range common.EnvWarnings {
		logger.Warn(warning)
	}

	logger.Info("启动 whats-api",
		"version", common.Version,
		"node_type", common.NodeType,
		"is_master", common.IsMasterNode,
		"port", common.Port,
		"sync_frequency_seconds", common.SyncFrequency,
		"shutdown_timeout_seconds", common.ShutdownTimeout,
	)

	// ---------- ② 数据层：唯一的致命初始化 ----------
	if err := model.InitDB(); err != nil {
		return err
	}
	if err := model.CheckSetup(); err != nil {
		return err
	}

	// ---------- ③ 配置必须在任何读配置的逻辑之前 ----------
	if err := model.InitOptionMap(); err != nil {
		return err
	}
	// 库里的配置可能覆盖了启动期的日志级别，这里对齐一次。
	// （InitOptionMap 里 applySideEffects 已经设过，这里是防御性的二次确认。）
	logger.SetLevel(logger.ParseLevel(system_setting.Get().LogLevel))

	// ---------- ④ 日志库（LOG_SQL_DSN 为空则复用主库）----------
	if err := model.InitLogDB(); err != nil {
		return err
	}

	// ---------- ⑤ 缓存：非致命，失败降级 ----------
	common.InitRedisClient()

	// ---------- ⑥ 后台任务 ----------
	// 全部接收 ctx：ctx 一取消它们就开始退出，停机时不会被写进已关闭的数据库。
	var wg sync.WaitGroup
	wg.Add(1)
	go func() {
		defer wg.Done()
		model.SyncOptions(ctx, time.Duration(common.SyncFrequency)*time.Second)
	}()

	// ---------- ⑦ HTTP ----------
	requests := cachex.NewCounter("requests", requestsCounterTTL)

	gin.SetMode(ginMode(common.GinMode))
	engine := gin.New()
	router.SetRouter(engine, router.Deps{Requests: requests, StartedAt: time.Now()})
	if opts.InjectRoutes != nil {
		opts.InjectRoutes(engine)
	}

	srv := &http.Server{
		Addr:    fmt.Sprintf(":%d", common.Port),
		Handler: engine,
		// ★ 刻意不设 ReadTimeout / WriteTimeout：
		// 流式响应（SSE）可能持续几分钟，写超时会把正常的流掐断。
		// 用 ReadHeaderTimeout 挡住慢速攻击，同时不影响响应时长。
		ReadHeaderTimeout: 10 * time.Second,
		IdleTimeout:       120 * time.Second,
	}

	serveErr := make(chan error, 1)
	go func() {
		logger.Info("HTTP 服务已启动", "addr", srv.Addr)
		if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
			serveErr <- err
			return
		}
		serveErr <- nil
	}()

	// ---------- ⑧ 等待退出条件 ----------
	var runErr error
	select {
	case <-ctx.Done():
		logger.Info("收到退出信号，开始优雅停机", "timeout_seconds", common.ShutdownTimeout)
	case err := <-serveErr:
		runErr = err
		if err == nil {
			logger.Info("HTTP 服务已自行退出")
		} else {
			logger.Error("HTTP 服务异常退出", "error", err)
		}
	}

	// ---------- ⑨ 优雅停机 ----------
	// 用独立的 ctx：此时上层 ctx 已经取消，不能再拿它当超时来源。
	shutdownCtx, cancelShutdown := context.WithTimeout(
		context.Background(), time.Duration(common.ShutdownTimeout)*time.Second)
	defer cancelShutdown()

	if err := srv.Shutdown(shutdownCtx); err != nil {
		// 超时不等于出错：可能是长流还没结束。记错误但继续收尾。
		logger.Error("优雅停机超时，仍有在途请求被中断", "error", err)
	}

	// 等后台 goroutine 退出**再**关库：反过来会让它们写到已关闭的连接上。
	wg.Wait()

	if err := model.CloseDB(); err != nil {
		logger.Warn("关闭数据库出错", "error", err)
	}
	if err := common.CloseRedis(); err != nil {
		logger.Warn("关闭 Redis 出错", "error", err)
	}

	logger.Info("已退出")
	return runErr
}

// ginMode 把环境变量映射成 gin 模式，未知值回退 release
// （回退到 debug 会让 gin 打印一堆警告，回退到 test 会静默一些错误日志）。
func ginMode(mode string) string {
	switch mode {
	case gin.DebugMode, gin.ReleaseMode, gin.TestMode:
		return mode
	default:
		return gin.ReleaseMode
	}
}

// Command optctl 直接读写 Option 表，绕过 HTTP。
//
// 为什么 M0 需要它：管理面的写接口是 M3 的事，而验收标准③要求验证
// "多实例改配置后能在 SYNC_FREQUENCY 内收敛"。要证这件事，就必须有一个
// **从进程外部**改配置的手段——否则测试改的是同一个进程的内存，等于没测同步。
//
// 它只做一层薄壳：解析参数 → 调 model.UpdateOption / OptionMap。
// 所有校验都在 model 层，因此它的行为与运行时完全一致（不会造出运行时拒绝的配置）。
//
// 用法：
//
//	optctl get  <key>
//	optctl set  <key> <json-value>
//	optctl list
//
// 值必须是 JSON 编码：字符串要带引号 —— optctl set system_setting.status_message '"你好"'
package main

import (
	"fmt"
	"io"
	"os"
	"sort"
	"strings"

	"github.com/yyyCode/whats-api/internal/common"
	"github.com/yyyCode/whats-api/internal/logger"
	"github.com/yyyCode/whats-api/internal/model"
)

func main() {
	if err := initOptctl(); err != nil {
		fmt.Fprintf(os.Stderr, "optctl 初始化失败: %v\n", err)
		os.Exit(1)
	}
	defer func() { _ = model.CloseDB() }()

	if err := run(os.Args[1:], os.Stdout); err != nil {
		fmt.Fprintf(os.Stderr, "optctl: %v\n", err)
		os.Exit(1)
	}
}

// initOptctl 走与运行时相同的初始化路径（环境变量 → 主库 → 配置装载），
// 保证工具看到的配置与进程看到的一致。
func initOptctl() error {
	common.InitEnv()
	logger.SetupLogger(common.LogLevel, false)

	if err := model.InitDB(); err != nil {
		return err
	}
	return model.InitOptionMap()
}

// run 是命令实现，与 main 分开是为了可测试：测试直接调 run，不必 exec 二进制。
func run(args []string, out io.Writer) error {
	if len(args) == 0 {
		return fmt.Errorf("缺少子命令\n%s", usage)
	}

	switch args[0] {
	case "get":
		if len(args) < 2 {
			return fmt.Errorf("get 需要 <key> 参数\n%s", usage)
		}
		return runGet(args[1], out)

	case "set":
		if len(args) < 3 {
			return fmt.Errorf("set 需要 <key> <json-value> 两个参数\n%s", usage)
		}
		return runSet(args[1], args[2])

	case "list":
		return runList(out)

	default:
		return fmt.Errorf("未知子命令 %q\n%s", args[0], usage)
	}
}

func runGet(key string, out io.Writer) error {
	value, ok := model.OptionMap[key]
	if !ok {
		return fmt.Errorf("未注册的配置项 %q", key)
	}
	fmt.Fprintln(out, value)
	return nil
}

func runSet(key, value string) error {
	if err := model.UpdateOption(key, value); err != nil {
		return err
	}
	// 回显写入后的值：让人确认"写进去的是什么"，而不是相信自己记得参数的引号。
	fmt.Printf("%s = %s\n", key, model.OptionMap[key])
	return nil
}

func runList(out io.Writer) error {
	keys := make([]string, 0, len(model.OptionMap))
	for key := range model.OptionMap {
		keys = append(keys, key)
	}
	sort.Strings(keys)

	for _, key := range keys {
		fmt.Fprintf(out, "%s = %s\n", key, model.OptionMap[key])
	}
	return nil
}

const usage = strings.TrimSpace(`
用法:
  optctl get  <key>
  optctl set  <key> <json-value>
  optctl list

值必须是 JSON 编码，字符串要带引号：
  optctl set system_setting.status_message '"你好"'
  optctl set system_setting.enable_access_log false
`)

RUN CGO_ENABLED=0 GOOS=linux go build \
      -trimpath \
      -ldflags="-s -w" \
      -o /out/whats-api .

FROM alpine:3.20

# tzdata：日志与账本按本地时区展示（存储仍是 unix 秒）。
RUN apk add --no-cache ca-certificates tzdata \
 && adduser -D -u 10001 app

WORKDIR /app
COPY --from=builder /out/whats-api /app/whats-api

# 数据落在卷里：容器重建不丢配置与库。

  CMD wget -qO- http://127.0.0.1:3000/api/status >/dev/null || exit 1

# 容器里本进程就是 PID 1。PID 1 默认会把未显式处理的信号忽略掉，
# 所以必须自己处理 SIGTERM（app.Run 收到 ctx 取消后做优雅停机）——

#!/usr/bin/env bash
# M0 验收脚本：驱动设计文档 §10.1 的四条验收标准。
#
# 判定三态（这是这个脚本最重要的一点）：
#   PASS            本机完整验证通过
#   NEEDS-EXTERNAL  本机环境不具备验证条件（无 docker / 无法投递 SIGTERM），
#                   需要你在 Linux 或 docker 环境复核 —— 而不是假装通过
#   FAIL            验证失败；只要有一个 FAIL，脚本退出码为 1
#
# 用法：
#   ./scripts/m0-acceptance.sh
#   BIN=/path/to/whats-api ./scripts/m0-acceptance.sh
set -uo pipefail

ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
BIN="${BIN:-$ROOT/whats-api}"
WORK="$(mktemp -d)"
PIDS=()
PASS=0
FAIL=0
NEED=0

cleanup() {
  for pid in "${PIDS[@]:-}"; do
    [ -n "$pid" ] && kill "$pid" 2>/dev/null
  done
  rm -rf "$WORK"
}
trap cleanup EXIT

pass() { printf '  \033[32mPASS\033[0m            %s\n' "$1"; PASS=$((PASS + 1)); }
fail() { printf '  \033[31mFAIL\033[0m            %s\n' "$1"; FAIL=$((FAIL + 1)); }
need() { printf '  \033[33mNEEDS-EXTERNAL\033[0m  %s\n' "$1"; NEED=$((NEED + 1)); }

is_windows() {
  case "$(uname -s)" in
    MINGW* | MSYS* | CYGWIN*) return 0 ;;
    *) return 1 ;;
  esac
}

# json_get <json> <key>，key 支持 "db.driver" 这种一层嵌套
json_get() {
  local json="$1" key="$2"
  if command -v jq >/dev/null 2>&1; then
    printf '%s' "$json" | jq -r --arg k "$key" '
      ($k | split(".")) as $p
      | if ($p | length) == 1 then .[$p[0]] else .[$p[0]][$p[1]] end
      | tostring' 2>/dev/null
    return
  fi
  # 没有 jq 时退化为取第一个同名键——本脚本只用它读顶层与 db/cache 一层嵌套，
  # 而 JSON 里这些键名在各自段内唯一。失败时脚本会打印完整 JSON 供人工核对。
  printf '%s' "$json" | tr -d '\n' \
    | sed -n "s/.*\"${key##*.}\"[[:space:]]*:[[:space:]]*\([^,}]*\).*/\1/p" \
    | tr -d '"' | head -1
}

# wait_status <port> <timeout_seconds>，成功则把 body 打到 stdout
wait_status() {
  local port="$1" timeout="${2:-20}" deadline body
  deadline=$(( $(date +%s) + timeout ))
  while [ "$(date +%s)" -lt "$deadline" ]; do
    if body="$(curl -fsS "http://127.0.0.1:$port/api/status" 2>/dev/null)"; then
      printf '%s' "$body"
      return 0
    fi
    sleep 0.3
  done
  return 1
}

# start_server <dir> [ENV=V ...] → 输出 pid
# 用 exec 让子 shell 被 env/二进制替换，于是 $! 就是二进制的 pid，kill 才有效。
start_server() {
  local dir="$1"
  shift
  ( cd "$dir" && exec env "$@" "$BIN" ) >"$dir/server.log" 2>&1 &
  echo $!
}

stop_server() {
  local pid="$1"
  [ -z "$pid" ] && return 0
  kill "$pid" 2>/dev/null
  wait "$pid" 2>/dev/null
}

echo "M0 验收开始（BIN=$BIN）"
echo

# ---------- 前置：构建二进制 ----------
if [ ! -x "$BIN" ]; then
  echo "构建二进制..."
  if ! ( cd "$ROOT" && go build -o "$BIN" . ); then
    echo "构建失败，无法继续" >&2
    exit 1
  fi
fi

OPTCTL="$WORK/optctl"
if ! ( cd "$ROOT" && go build -o "$OPTCTL" ./cmd/optctl ); then
  echo "构建 optctl 失败，无法继续" >&2
  exit 1
fi

# ---------- ① 零环境变量启动（SQLite）----------
# 在一个空目录里启动：确保不会读到仓库根的 .env（那会让"零环境变量"名不副实）。
echo "① 零环境变量启动（SQLite）"
D1="$WORK/t1"
mkdir -p "$D1"
PID1="$(start_server "$D1")"
PIDS+=("$PID1")

if BODY1="$(wait_status 3000 25)"; then
  DRIVER="$(json_get "$BODY1" db.driver)"
  DSNKIND="$(json_get "$BODY1" db.dsn_kind)"
  if [ "$DRIVER" = "sqlite" ]; then
    pass "① 零环境变量启动成功，db.driver=sqlite，dsn_kind=$DSNKIND"
  else
    fail "① 启动了但 db.driver=$DRIVER（期望 sqlite）"
  fi
  if [ -f "$D1/whats-api.db" ]; then
    pass "① SQLite 文件按默认名建在启动目录（whats-api.db）"
  else
    fail "① 未在启动目录看到 whats-api.db（$(ls -a "$D1")）"
  fi
else
  fail "① 不配任何环境变量时未能启动（端口 3000 可能被占用；日志见 $D1/server.log）"
  echo "     最后几行日志："
  tail -n 10 "$D1/server.log" 2>/dev/null | sed 's/^/       /'
fi
stop_server "$PID1"

# ---------- ①' Docker 形态 ----------
if command -v docker >/dev/null 2>&1 && docker info >/dev/null 2>&1; then
  if ( cd "$ROOT" && docker build -q -t whats-api:m0 . >"$WORK/docker-build.log" 2>&1 ); then
    CID="$(docker run -d -p 3900:3000 whats-api:m0 2>/dev/null)"
    if DBODY="$(wait_status 3900 40)"; then
      if [ "$(json_get "$DBODY" db.driver)" = "sqlite" ]; then
        pass "① Docker 形态：容器零环境变量启动，db.driver=sqlite"
      else
        fail "① Docker 形态：驱动不符（$(json_get "$DBODY" db.driver)）"
      fi
    else
      fail "① Docker 形态：容器起来了但 /api/status 不响应"
    fi
    [ -n "${CID:-}" ] && docker rm -f "$CID" >/dev/null 2>&1
  else
    fail "① Docker 形态：docker build 失败（见 $WORK/docker-build.log）"
  fi
else
  need "① Docker 形态：本机无可用 docker，请在 Linux/docker 环境跑 docker compose up 复核"
fi

# ---------- ② 拿掉 Redis 也能启动 ----------
echo
echo "② 拿掉 Redis 也能启动（降级为内存缓存）"
D2="$WORK/t2"
mkdir -p "$D2"
# 故意配一个不可达的地址（端口 1 保留且无监听）：必须启动成功且降级。
PID2="$(start_server "$D2" "PORT=3001" "REDIS_CONN_STRING=127.0.0.1:1" "SQL_DSN=local:$D2/whats-api.db")"
PIDS+=("$PID2")

if BODY2="$(wait_status 3001 25)"; then
  BACKEND="$(json_get "$BODY2" cache.backend)"
  CONFIGURED="$(json_get "$BODY2" cache.redis_configured)"
  if [ "$BACKEND" = "memory" ]; then
    pass "② Redis 不可达时仍启动，cache.backend=memory（redis_configured=$CONFIGURED）"
  else
    fail "② cache.backend=$BACKEND，期望 memory"
  fi
  if [ "$CONFIGURED" = "true" ]; then
    pass "② 状态接口如实反映「配置了 Redis 但降级了」（redis_configured=true）"
  else
    fail "② redis_configured=$CONFIGURED，期望 true（应反映配置存在但未连上）"
  fi
  if grep -qi "降级" "$D2/server.log"; then
    pass "② 启动日志里有降级告警（不是静默忽略）"
  else
    fail "② 启动日志里没有降级告警"
  fi
else
  fail "② 配上不可达的 Redis 后启动失败（不应如此：缓存必须是非致命的）"
  tail -n 10 "$D2/server.log" 2>/dev/null | sed 's/^/       /'
fi
stop_server "$PID2"

# ---------- ③ 配置热更新与多实例收敛 ----------
echo
echo "③ 改配置：单实例立即生效 + 多实例 SYNC_FREQUENCY 内收敛"
D3="$WORK/t3"
mkdir -p "$D3"
SHARED_DB="local:$D3/shared.db"
MSG="脚本改的-$$"

PID3A="$(start_server "$D3" "PORT=3002" "SQL_DSN=$SHARED_DB" "SYNC_FREQUENCY=5" "NODE_NAME=node-a" "LOG_LEVEL=info")"
PIDS+=("$PID3A")
BODY_A="$(wait_status 3002 25)" || true

PID3B="$(start_server "$D3" "PORT=3003" "SQL_DSN=$SHARED_DB" "SYNC_FREQUENCY=5" "NODE_NAME=node-b" "LOG_LEVEL=info")"
PIDS+=("$PID3B")
BODY_B="$(wait_status 3003 25)" || true

if [ -z "${BODY_A:-}" ] || [ -z "${BODY_B:-}" ]; then
  fail "③ 两个实例未能同时启动（A=${BODY_A:0:40} B=${BODY_B:0:40}）"
else
  pass "③ 两个实例共享同一个 SQLite 库启动成功"

  BEFORE_B="$(json_get "$BODY_B" status_message)"

  # ★ 从进程外改配置：这正是"多实例同步"与"改内存"的区别所在
  if env "SQL_DSN=$SHARED_DB" "$OPTCTL" set system_setting.status_message "\"$MSG\"" >"$WORK/optctl.log" 2>&1; then
    pass "③ optctl 从进程外写入配置成功"
  else
    fail "③ optctl 写入失败（见 $WORK/optctl.log）"
  fi

  # 轮询实例 B：SYNC_FREQUENCY=5，给它 5 倍余量
  CONVERGED=""
  for _ in $(seq 1 25); do
    if CURRENT="$(json_get "$(curl -fsS http://127.0.0.1:3003/api/status 2>/dev/null)" status_message)" \
       && [ "$CURRENT" = "$MSG" ]; then
      CONVERGED="yes"
      break
    fi
    sleep 1
  done

  if [ -n "$CONVERGED" ]; then
    pass "③ 实例 B 在同步周期内收敛到新值（改前 $BEFORE_B → 改后 $MSG）"
  else
    fail "③ 实例 B 未收敛（期望 $MSG，实际 $(json_get "$(curl -fsS http://127.0.0.1:3003/api/status 2>/dev/null)" status_message)）"
  fi

  A_NOW="$(json_get "$(curl -fsS http://127.0.0.1:3002/api/status 2>/dev/null)" status_message)"
  if [ "$A_NOW" = "$MSG" ]; then
    pass "③ 实例 A 同样收敛"
  else
    fail "③ 实例 A 未收敛（实际 $A_NOW）"
  fi
fi
stop_server "$PID3A"
stop_server "$PID3B"

# ---------- ④ SIGTERM 优雅停机 ----------
echo
echo "④ SIGTERM 优雅停机"
if is_windows; then
  need "④ 优雅停机：Windows 无法向原生进程投递 SIGTERM，请在 Linux/docker 环境复核"
  echo "     本机可验证的等价路径已由单元测试覆盖："
  echo "       go test -race ./internal/app/ -run TestRunDrainsInFlightRequest"
else
  D4="$WORK/t4"
  mkdir -p "$D4"
  PID4="$(start_server "$D4" "PORT=3004" "SQL_DSN=local:$D4/whats-api.db")"
  if wait_status 3004 25 >/dev/null; then
    kill -TERM "$PID4"
    # 给足时间：停机超时默认 120 秒，但正常情况下应当秒级返回
    WAITED=0
    while kill -0 "$PID4" 2>/dev/null && [ "$WAITED" -lt 30 ]; do
      sleep 0.5
      WAITED=$((WAITED + 1))
    done

    if kill -0 "$PID4" 2>/dev/null; then
      fail "④ 收到 SIGTERM 后 15 秒内没有退出"
      kill -9 "$PID4" 2>/dev/null
    else
      wait "$PID4"
      CODE=$?
      if [ "$CODE" -eq 0 ]; then
        pass "④ 收到 SIGTERM 后正常退出（退出码 0）"
      else
        fail "④ 退出码 = $CODE，期望 0"
      fi
      if grep -q "已退出" "$D4/server.log"; then
        pass "④ 日志里有完整的停机序列（收到退出信号 → 已退出）"
      else
        fail "④ 日志里没有停机序列"
      fi
    fi
  else
    fail "④ 实例未能启动，无法测试停机"
  fi
fi

# ---------- 汇总 ----------
echo
echo "----------------------------------------"
printf 'PASS=%d  FAIL=%d  NEEDS-EXTERNAL=%d\n' "$PASS" "$FAIL" "$NEED"
if [ "$NEED" -gt 0 ]; then
  echo "说明：NEEDS-EXTERNAL 表示本机环境无法验证，需在 Linux/docker 环境复核（不是通过）。"
fi
if [ "$FAIL" -gt 0 ]; then
  echo "结论：验收未通过"
  exit 1
fi
echo "结论：本机可验证部分全部通过"
```

```markdown
## 功能描述 (Feature Description)

落地 M0 技术骨架：一个能启动、能优雅停机、能热更新配置、能探活的空壳服务，
目录结构与依赖方向就此定型。本次不包含任何转发、计费、管理面 CRUD 与前端。

- `GET /api/status` 是唯一的业务端点：返回版本、节点类型、主库/日志库连通性、
  缓存后端、已注册配置域、请求总数，以及配置项 `status_message` / `log_level` 等
  （改配置后这些字段立刻变化，是验收标准③的观察窗）。
- 不配任何环境变量即可启动（SQLite 文件 + 进程内缓存）；配 `SQL_DSN` 可切
  MySQL / PostgreSQL，配 `LOG_SQL_DSN` 可把日志库分离出去。
- `cmd/optctl`：`get` / `set` / `list` 直接读写配置表，用于不重启改配置与多实例收敛验证。
- 四条验收标准由 `scripts/m0-acceptance.sh` 驱动，输出 PASS / FAIL / NEEDS-EXTERNAL。

## 实现思路 (Implementation Approach)

- **配置系统**：一个结构体 + json tag + `Register` 同时定义类型、默认值、持久化 key 与文档，
  避免多份定义漂移；业务代码只读 `atomic.Pointer` 快照，因此热路径无锁、且不会读到中间态
  （直接读被反射写入的结构体会是 data race）。写入只有 `UpdateOption` 一条路径：
  校验 → 落库 → 改内存/快照/副作用，顺序不可颠倒。
- **数据层**：GORM 一份代码兼容三库，方言差异（保留字列、布尔字面量、行锁）收进函数；
  迁移只在 `NODE_TYPE=master` 执行且幂等；`logs` 表从一开始就落在独立日志库，
  让"日志与业务分库"这条接缝当下就可验证。SQLite 用纯 Go 驱动，因此 `CGO_ENABLED=0` 可静态编译。
- **运行时**：启动顺序把"配置装载"排在所有读配置的逻辑之前；除了主库连接，
  其余初始化（.env、Redis、日志库）失败一律降级不阻断。停机做成可被 ctx 取消驱动的
  `app.Run(ctx)`——本机（Windows）无法投递 SIGTERM，做成函数才能真实测到
  "在途请求跑完 → 后台 goroutine 退出 → 关库"。
- **HTTP**：gin 三层路由（`/api` 管理面、`/v1` 转发面预留、NoRoute 兜底）。
  未实现的 `/v1/*` 与 `/api/*` 一律返回 JSON 404，不被将来的 SPA 兜底吞成 HTML。
  两个面的错误形状从一开始就分开。
- **明确不做**：转发端点、计费、管理面 CRUD、前端、渠道选路、指标与任务框架（M1-M5）。

1. `go test -race ./...` —— 全绿。重点覆盖：数值安全的三档饱和语义、
   配置写入路径（含"失败写入不留任何痕迹"）、配置快照的并发读、

   `logs` 落在独立日志库、优雅停机排水与 DB 关闭。
2. `./scripts/m0-acceptance.sh` —— 驱动四条验收标准，输出见
